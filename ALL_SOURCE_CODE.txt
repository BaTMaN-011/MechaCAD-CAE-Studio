"""Desktop preferences and OS-backed storage for database credentials."""
from __future__ import annotations

from dataclasses import dataclass
from PySide6.QtCore import QSettings


@dataclass(frozen=True)
class DatabaseConfig:
    host: str
    port: int
    database: str
    username: str
    password: str
    use_tls: bool = True
    verify_tls: bool = True
    ssl_ca: str = ""


class AppSettings:
    """Store non-secret preferences in QSettings and secrets in keyring."""
    SERVICE_NAME = "MechaCAD-CAE-Studio"

    def __init__(self) -> None:
        self._settings = QSettings("MechaCADSchoolProject", "MechaCAD-CAE-Studio")

    @staticmethod
    def _keyring():
        try:
            import keyring
            return keyring
        except Exception:
            return None

    @staticmethod
    def credential_key(host: str, database: str, username: str) -> str:
        return f"{username}@{host}/{database}"

    def load_connection(self) -> dict[str, object] | None:
        host = str(self._settings.value("mysql/host", "")).strip()
        database = str(self._settings.value("mysql/database", "")).strip()
        username = str(self._settings.value("mysql/username", "")).strip()
        if not host or not database or not username:
            return None
        keyring = self._keyring()
        password = ""
        if keyring:
            try:
                password = keyring.get_password(
                    self.SERVICE_NAME,
                    self.credential_key(host, database, username),
                ) or ""
            except Exception:
                password = ""
        return {
            "host": host,
            "port": int(self._settings.value("mysql/port", 3306)),
            "database": database,
            "username": username,
            "password": password,
            "use_tls": self._settings.value("mysql/use_tls", True, type=bool),
            "verify_tls": self._settings.value("mysql/verify_tls", True, type=bool),
            "ssl_ca": str(self._settings.value("mysql/ssl_ca", "")),
        }

    def save_connection(self, config: DatabaseConfig, remember_password: bool) -> None:
        self._settings.setValue("mysql/host", config.host)
        self._settings.setValue("mysql/port", config.port)
        self._settings.setValue("mysql/database", config.database)
        self._settings.setValue("mysql/username", config.username)
        self._settings.setValue("mysql/use_tls", config.use_tls)
        self._settings.setValue("mysql/verify_tls", config.verify_tls)
        self._settings.setValue("mysql/ssl_ca", config.ssl_ca)
        keyring = self._keyring()
        if not keyring:
            return
        key = self.credential_key(config.host, config.database, config.username)
        try:
            if remember_password and config.password:
                keyring.set_password(self.SERVICE_NAME, key, config.password)
            else:
                try:
                    keyring.delete_password(self.SERVICE_NAME, key)
                except Exception:
                    pass
        except Exception:
            # The app remains usable; the user can enter the password again next time.
            pass

class DatabaseManager:
    """Keeps one authenticated connection to a local or remote MySQL database."""

    def __init__(self) -> None:
        self._connection = None
        self.config: DatabaseConfig | None = None

    @property
    def connected(self) -> bool:
        return self._connection is not None and self._connection.is_connected()

    def disconnect(self) -> None:
        if self._connection is not None:
            try:
                self._connection.close()
            finally:
                self._connection = None
                self.config = None

    def connect(self, config: DatabaseConfig) -> None:
        """Connect, check access, install tables, and add starter materials."""
        self.disconnect()
        options: dict[str, Any] = {
            "host": config.host,
            "port": config.port,
            "database": config.database,
            "user": config.username,
            "password": config.password,
            "connection_timeout": 8,
            "autocommit": False,
            "use_pure": True,
        }
        if config.use_tls:
            if config.ssl_ca:
                options["ssl_ca"] = config.ssl_ca
            options.update({
                "ssl_disabled": False,
                "ssl_verify_cert": config.verify_tls,
                "ssl_verify_identity": config.verify_tls,
            })
        else:
            options["ssl_disabled"] = True
        try:
            self._connection = mysql.connector.connect(**options)
            self.config = config
            self._create_schema()
        except MySQLError:
            self.disconnect()
            raise

    def _ready_connection(self):
        if self._connection is None:
            raise RuntimeError("Connect to a MySQL database first.")
        try:
            self._connection.ping(reconnect=True, attempts=2, delay=1)
        except MySQLError:
            self.disconnect()
            raise
        return self._connection

    def _create_schema(self) -> None:
        connection = self._ready_connection()
        cursor = connection.cursor()
        try:
            for statement in SCHEMA:
                cursor.execute(statement)
            connection.commit()
        except Exception:
            connection.rollback()
            raise
        finally:
            cursor.close()
        self._seed_materials()

    def _seed_materials(self) -> None:
        connection = self._ready_connection()
        cursor = connection.cursor()
        try:
            cursor.executemany(
                """INSERT IGNORE INTO materials
                (name, category, youngs_modulus_mpa, poisson_ratio, density_kg_m3,
                 yield_strength_mpa, notes) VALUES (%s, %s, %s, %s, %s, %s, %s)""",
                STARTER_MATERIALS,
            )
            connection.commit()
        except Exception:
            connection.rollback()
            raise
        finally:
            cursor.close()

    def _query(self, sql: str, params: tuple = (), *, fetch: bool = False):
        connection = self._ready_connection()
        cursor = connection.cursor(dictionary=True)
        try:
            cursor.execute(sql, params)
            result = cursor.fetchall() if fetch else cursor.lastrowid
            if not fetch:
                connection.commit()
            return result
        except Exception:
            connection.rollback()
            raise
        finally:
            cursor.close()

    def list_projects(self) -> list[dict[str, Any]]:
        return self._query(
            "SELECT id, name, description, created_at, updated_at FROM projects ORDER BY updated_at DESC",
            fetch=True,
        )

    def create_project(self, name: str, description: str) -> int:
        return int(self._query(
            "INSERT INTO projects (name, description) VALUES (%s, %s)",
            (name.strip(), description.strip()),
        ))

    def update_project(self, project_id: int, name: str, description: str) -> None:
        self._query(
            "UPDATE projects SET name = %s, description = %s WHERE id = %s",
            (name.strip(), description.strip(), project_id),
        )

    def delete_project(self, project_id: int) -> None:
        self._query("DELETE FROM projects WHERE id = %s", (project_id,))

    def project_counts(self) -> dict[str, int]:
        rows = self._query(
            """SELECT
                (SELECT COUNT(*) FROM projects) AS projects,
                (SELECT COUNT(*) FROM designs) AS designs,
                (SELECT COUNT(*) FROM analysis_runs) AS analyses,
                (SELECT COUNT(*) FROM materials) AS materials,
                (SELECT COUNT(*) FROM design_checks) AS checks""",
            fetch=True,
        )
        return {key: int(value) for key, value in rows[0].items()}

    def list_materials(self, search: str = "") -> list[dict[str, Any]]:
        pattern = f"%{search.strip()}%"
        return self._query(
            """SELECT id, name, category, youngs_modulus_mpa, poisson_ratio,
                      density_kg_m3, yield_strength_mpa, notes
               FROM materials
               WHERE name LIKE %s OR category LIKE %s
               ORDER BY category, name""",
            (pattern, pattern),
            fetch=True,
        )

    def add_material(
        self,
        name: str,
        category: str,
        youngs_modulus_mpa: float,
        poisson_ratio: float,
        density_kg_m3: float,
        yield_strength_mpa: float,
        notes: str,
    ) -> int:
        return int(self._query(
            """INSERT INTO materials
               (name, category, youngs_modulus_mpa, poisson_ratio, density_kg_m3, yield_strength_mpa, notes)
               VALUES (%s, %s, %s, %s, %s, %s, %s)""",
            (name.strip(), category.strip(), youngs_modulus_mpa, poisson_ratio,
             density_kg_m3, yield_strength_mpa, notes.strip()),
        ))

    def update_material(
        self, material_id: int, name: str, category: str, youngs_modulus_mpa: float,
        poisson_ratio: float, density_kg_m3: float, yield_strength_mpa: float, notes: str,
    ) -> None:
        self._query(
            """UPDATE materials SET name = %s, category = %s, youngs_modulus_mpa = %s,
               poisson_ratio = %s, density_kg_m3 = %s, yield_strength_mpa = %s, notes = %s
               WHERE id = %s""",
            (name.strip(), category.strip(), youngs_modulus_mpa, poisson_ratio,
             density_kg_m3, yield_strength_mpa, notes.strip(), material_id),
        )

    def list_components(self, project_id: int) -> list[dict[str, Any]]:
        return self._query(
            """SELECT c.id, c.part_number, c.name, c.material_id, c.quantity, c.volume_mm3,
                      c.unit_mass_kg, c.notes, m.name AS material_name
               FROM components c LEFT JOIN materials m ON m.id = c.material_id
               WHERE c.project_id = %s ORDER BY c.part_number, c.name""",
            (project_id,),
            fetch=True,
        )

    def save_component(
        self, project_id: int, part_number: str, name: str, material_id: int | None,
        quantity: int, volume_mm3: float, unit_mass_kg: float, notes: str,
        component_id: int | None = None,
    ) -> int:
        values = (part_number.strip(), name.strip(), material_id, quantity, volume_mm3,
                  unit_mass_kg, notes.strip())
        if component_id is not None:
            self._query(
                """UPDATE components SET part_number = %s, name = %s, material_id = %s,
                   quantity = %s, volume_mm3 = %s, unit_mass_kg = %s, notes = %s
                   WHERE id = %s AND project_id = %s""",
                values + (component_id, project_id),
            )
            return component_id
        return int(self._query(
            """INSERT INTO components
               (project_id, part_number, name, material_id, quantity, volume_mm3, unit_mass_kg, notes)
               VALUES (%s, %s, %s, %s, %s, %s, %s, %s)""",
            (project_id,) + values,
        ))

    def delete_component(self, component_id: int, project_id: int) -> None:
        self._query("DELETE FROM components WHERE id = %s AND project_id = %s", (component_id, project_id))

    def load_design(self, project_id: int, name: str = "Main Sketch") -> dict[str, Any] | None:
        rows = self._query(
            "SELECT id, name, entities_json FROM designs WHERE project_id = %s AND name = %s LIMIT 1",
            (project_id, name),
            fetch=True,
        )
        if not rows:
            return None
        row = rows[0]
        try:
            row["entities"] = json.loads(row.pop("entities_json"))
        except (TypeError, json.JSONDecodeError):
            row["entities"] = []
        return row

    def save_design(self, project_id: int, entities: list[dict[str, Any]], name: str = "Main Sketch") -> int:
        payload = json.dumps(entities, ensure_ascii=False)
        existing = self.load_design(project_id, name)
        if existing:
            self._query(
                "UPDATE designs SET entities_json = %s WHERE id = %s",
                (payload, existing["id"]),
            )
            return int(existing["id"])
        return int(self._query(
            "INSERT INTO designs (project_id, name, entities_json) VALUES (%s, %s, %s)",
            (project_id, name, payload),
        ))

    def save_analysis(self, project_id: int, model: dict, results: dict, name: str) -> int:
        return int(self._query(
            """INSERT INTO analysis_runs (project_id, name, model_json, results_json)
               VALUES (%s, %s, %s, %s)""",
            (project_id, name.strip(), json.dumps(model), json.dumps(results)),
        ))

    def list_analyses(self, project_id: int) -> list[dict[str, Any]]:
        return self._query(
            """SELECT id, name, created_at, results_json FROM analysis_runs
               WHERE project_id = %s ORDER BY created_at DESC""",
            (project_id,),
            fetch=True,
        )

    def save_design_check(
        self, project_id: int, name: str, check_type: str, inputs: dict, results: dict
    ) -> int:
        return int(self._query(
            """INSERT INTO design_checks
               (project_id, name, check_type, inputs_json, results_json)
               VALUES (%s, %s, %s, %s, %s)""",
            (project_id, name.strip(), check_type, json.dumps(inputs), json.dumps(results)),
        ))

    def list_design_checks(self, project_id: int) -> list[dict[str, Any]]:
        return self._query(
            """SELECT id, name, check_type, created_at, inputs_json, results_json
               FROM design_checks WHERE project_id = %s ORDER BY created_at DESC""",
            (project_id,),
            fetch=True,
        )

"""Assembly roll-up metrics and mass completeness notes."""
from __future__ import annotations

from dataclasses import dataclass
from typing import Any


@dataclass(frozen=True)
class AssemblySummary:
    line_items: int
    total_quantity: int
    total_mass_kg: float
    lines_with_material: int
    lines_with_mass: int
    warnings: tuple[str, ...]


def summarize_assembly(components: list[dict[str, Any]]) -> AssemblySummary:
    """Summarize BOM mass and identify rows that need material review."""
    total_quantity = 0
    total_mass = 0.0
    material_rows = 0
    mass_rows = 0
    warnings = []
    for component in components:
        quantity = int(component.get("quantity", 0))
        mass_each = float(component.get("unit_mass_kg", 0) or 0)
        total_quantity += quantity
        total_mass += quantity * mass_each
        if component.get("material_id") is not None:
            material_rows += 1
        if mass_each > 0:
            mass_rows += 1
        elif quantity > 0:
            warnings.append(f"{component.get('part_number', 'Part')} has no estimated unit mass.")
    if components and material_rows < len(components):
        warnings.append("Some component rows do not reference a material.")
    return AssemblySummary(
        line_items=len(components),
        total_quantity=total_quantity,
        total_mass_kg=total_mass,
        lines_with_material=material_rows,
        lines_with_mass=mass_rows,
        warnings=tuple(warnings),
    )

"""Euler-Bernoulli beam formulas for common classroom load cases."""
from __future__ import annotations

from dataclasses import dataclass


@dataclass(frozen=True)
class BeamResult:
    support_model: str
    load_model: str
    maximum_moment_nmm: float
    maximum_deflection_mm: float
    maximum_bending_stress_mpa: float
    yield_safety_factor: float | None
    deflection_ratio: float
    assumptions: tuple[str, ...]


class BeamInputError(ValueError):
    pass


def calculate_beam(
    support_model: str,
    load_model: str,
    length_mm: float,
    elastic_modulus_mpa: float,
    second_moment_mm4: float,
    extreme_fibre_mm: float,
    *,
    point_load_n: float = 0.0,
    distributed_load_n_per_mm: float = 0.0,
    yield_strength_mpa: float | None = None,
) -> BeamResult:
    """Compute a maximum response for a uniform prismatic beam.

    Supported cases: simply supported with a centered point load or full-span
    uniform load; cantilever with an end point load or full-span uniform load.
    The result uses small-deflection, linear-elastic beam theory.
    """
    if support_model not in {"Simply supported", "Cantilever"}:
        raise BeamInputError("Choose a supported support model.")
    if load_model not in {"Point load", "Uniform load"}:
        raise BeamInputError("Choose a supported load model.")
    if min(length_mm, elastic_modulus_mpa, second_moment_mm4, extreme_fibre_mm) <= 0:
        raise BeamInputError("Length, modulus, section inertia, and fibre distance must be positive.")
    if yield_strength_mpa is not None and yield_strength_mpa <= 0:
        raise BeamInputError("Yield strength must be positive when supplied.")
    length = float(length_mm)
    modulus = float(elastic_modulus_mpa)
    inertia = float(second_moment_mm4)
    if load_model == "Point load":
        if point_load_n <= 0:
            raise BeamInputError("Enter a positive point load.")
        if support_model == "Simply supported":
            moment = point_load_n * length / 4
            deflection = point_load_n * length**3 / (48 * modulus * inertia)
        else:
            moment = point_load_n * length
            deflection = point_load_n * length**3 / (3 * modulus * inertia)
    else:
        if distributed_load_n_per_mm <= 0:
            raise BeamInputError("Enter a positive distributed load in N/mm.")
        if support_model == "Simply supported":
            moment = distributed_load_n_per_mm * length**2 / 8
            deflection = 5 * distributed_load_n_per_mm * length**4 / (384 * modulus * inertia)
        else:
            moment = distributed_load_n_per_mm * length**2 / 2
            deflection = distributed_load_n_per_mm * length**4 / (8 * modulus * inertia)

    stress = moment * extreme_fibre_mm / inertia
    safety = yield_strength_mpa / stress if yield_strength_mpa and stress > 0 else None
    return BeamResult(
        support_model=support_model,
        load_model=load_model,
        maximum_moment_nmm=moment,
        maximum_deflection_mm=deflection,
        maximum_bending_stress_mpa=stress,
        yield_safety_factor=safety,
        deflection_ratio=(length / deflection) if deflection > 0 else float("inf"),
        assumptions=(
            "Euler-Bernoulli beam theory; linear elastic, prismatic member.",
            "Point load acts at midspan for simply supported beams and at the free end for cantilevers.",
            "Uniform load covers the full span. Shear deformation, self-weight, and local instability are omitted.",
        ),
    )

"""Time-domain simulation of a position-controlled permanent-magnet DC motor."""
from __future__ import annotations

    rise_time_s: float | None
    settling_time_s: float | None
    overshoot_percent: float
    steady_state_error_deg: float

class ControlInputError(ValueError):

def simulate_dc_motor_position(
    target_deg: float,
    supply_voltage_v: float,
    duration_s: float,
    time_step_s: float,
    resistance_ohm: float,
    inductance_h: float,
    torque_constant_nm_per_a: float,
    back_emf_v_per_rad_s: float,
    inertia_kg_m2: float,

    kp: float,
    ki: float,
    kd: float,
    load_torque_nm: float = 0.0,

    """Simulate a DC motor and saturated PID voltage controller using Euler steps.

    Electrical and rotational parameters use SI units. The controller regulates
    shaft position in radians, and its gains produce volts. Current and speed
    are integrated with a fixed, user-selected time step.
    """
    positive = {
        "target angle": target_deg, "supply voltage": supply_voltage_v,
        "duration": duration_s, "time step": time_step_s, "resistance": resistance_ohm,
        "inductance": inductance_h, "torque constant": torque_constant_nm_per_a,
        "back-emf constant": back_emf_v_per_rad_s, "rotor inertia": inertia_kg_m2,
    }
    invalid = [name for name, value in positive.items() if value <= 0]
    if invalid:
        raise ControlInputError("Values must be positive: " + ", ".join(invalid))
    if min(viscous_friction_nm_per_rad_s, kp, ki, kd, load_torque_nm) < 0:
        raise ControlInputError("Friction, PID gains, and opposing load torque cannot be negative.")
    steps = int(math.ceil(duration_s / time_step_s))
    if steps > 50000:
        raise ControlInputError("Use a time step that produces no more than 50,000 simulation steps.")
    if time_step_s > min(inductance_h / resistance_ohm, math.sqrt(inertia_kg_m2 / max(kp, 0.01))) * 0.2:
        raise ControlInputError("Time step is too large for these motor parameters; use a smaller step.")

    target = math.radians(target_deg)
    angle, speed, current = 0.0, 0.0, 0.0
    integral = 0.0
    times, positions, currents, voltages = [], [], [], []
    saturated_steps = 0
    max_current = 0.0

    for step in range(steps + 1):
        time_value = min(step * time_step_s, duration_s)
        error = target - angle
        derivative = -speed
        candidate_integral = integral + error * time_step_s
        raw_voltage = kp * error + ki * candidate_integral + kd * derivative
        voltage = max(-supply_voltage_v, min(supply_voltage_v, raw_voltage))
        saturated = abs(raw_voltage - voltage) > 1e-12
        if not saturated or (raw_voltage > supply_voltage_v and error < 0) or (raw_voltage < -supply_voltage_v and error > 0):
            integral = candidate_integral
            raw_voltage = kp * error + ki * integral + kd * derivative
            voltage = max(-supply_voltage_v, min(supply_voltage_v, raw_voltage))
        if saturated:
            saturated_steps += 1

        times.append(time_value)
        positions.append(math.degrees(angle))
        currents.append(current)
        voltages.append(voltage)
        max_current = max(max_current, abs(current))
        if step == steps:
            break
        current_rate = (voltage - resistance_ohm * current - back_emf_v_per_rad_s * speed) / inductance_h
        acceleration = (
            torque_constant_nm_per_a * current - viscous_friction_nm_per_rad_s * speed - load_torque_nm
        ) / inertia_kg_m2
        current += current_rate * time_step_s
        speed += acceleration * time_step_s
        angle += speed * time_step_s

    rise_time = next((t for t, angle_deg in zip(times, positions) if angle_deg >= 0.9 * target_deg), None)
    tolerance = max(0.5, target_deg * 0.02)
    settling_time = None
    remains_settled = True
    for index in range(len(positions) - 1, -1, -1):
        remains_settled = remains_settled and abs(positions[index] - target_deg) <= tolerance
        if remains_settled:
            settling_time = times[index]
    peak = max(positions, default=0.0)
    overshoot = max(0.0, (peak - target_deg) / target_deg * 100.0)
    return MotorSimulation(
        time_s=times,
        position_deg=positions,
        current_a=currents,
        voltage_v=voltages,
        target_deg=target_deg,
        rise_time_s=rise_time,
        settling_time_s=settling_time,
        overshoot_percent=overshoot,
        steady_state_error_deg=target_deg - positions[-1],
        peak_current_a=max_current,
        saturation_percent=100 * saturated_steps / max(steps, 1),
    )

"""Idealized multistage spur-gear ratio and output estimate."""
from __future__ import annotations

from dataclasses import dataclass
from math import pi


@dataclass(frozen=True)
class GearStage:
    driver_teeth: int
    driven_teeth: int
    efficiency: float = 0.95


@dataclass(frozen=True)
class GearTrainResult:
    total_ratio: float
    total_efficiency: float
    output_speed_rpm: float
    output_torque_nm: float
    output_power_w: float
    stages: int


class GearInputError(ValueError):
    pass


def calculate_gear_train(input_torque_nm: float, input_speed_rpm: float, stages: list[GearStage]) -> GearTrainResult:
    """Calculate an approximate speed/torque change from simple gear stages.

    Ratio is defined as driver speed divided by driven speed. Intermeshing
    external gears reverse direction each stage; direction is omitted here.
    """
    if input_torque_nm <= 0 or input_speed_rpm <= 0:
        raise GearInputError("Input torque and speed must be positive.")
    if not stages:
        raise GearInputError("Add at least one gear stage.")
    ratio = 1.0
    efficiency = 1.0
    for index, stage in enumerate(stages, start=1):
        if stage.driver_teeth <= 0 or stage.driven_teeth <= 0:
            raise GearInputError(f"Stage {index} tooth counts must be positive.")
        if not 0 < stage.efficiency <= 1:
            raise GearInputError(f"Stage {index} efficiency must be above 0 and at most 1.")
        ratio *= stage.driven_teeth / stage.driver_teeth
        efficiency *= stage.efficiency
    output_speed = input_speed_rpm / ratio
    output_torque = input_torque_nm * ratio * efficiency
    output_power = output_torque * 2 * pi * output_speed / 60
    return GearTrainResult(ratio, efficiency, output_speed, output_torque, output_power, len(stages))

def entity_metrics(entity: dict) -> dict[str, float | str]:
    kind = str(entity.get("kind", "rectangle")).lower()

    if width <= 0 or height <= 0:
        raise ValueError("Sketch entity width and height must be positive.")
    if kind == "circle":
        radius = width / 2
        return {
            "kind": "circle", "area_mm2": math.pi * radius**2,
            "perimeter_mm": 2 * math.pi * radius,
            "diameter_mm": width,
        }
    if kind != "rectangle":
        raise ValueError(f"Unsupported sketch entity: {kind}")
    return {
        "kind": "rectangle", "area_mm2": width * height,
        "perimeter_mm": 2 * (width + height), "width_mm": width, "height_mm": height,
    }

    metrics = [entity_metrics(item) for item in entities]
    return {
        "entity_count": float(len(metrics)),

def export_dxf(entities: list[dict]) -> str:
    """Export rectangles as four LINE entities and circles as CIRCLE entities."""
    lines = [
        "0", "SECTION", "2", "HEADER", "9", "$INSUNITS", "70", "4", "0", "ENDSEC",
        "0", "SECTION", "2", "ENTITIES",
    ]

    def number(value: float) -> str:
        return f"{float(value):.6f}".rstrip("0").rstrip(".") or "0"

    def add_line(x1: float, y1: float, x2: float, y2: float) -> None:
        lines.extend((
            "0", "LINE", "8", "SKETCH",
            "10", number(x1), "20", number(y1), "30", "0",
            "11", number(x2), "21", number(y2), "31", "0",
        ))

    for entity in entities:
        x, y = float(entity.get("x", 0)), float(entity.get("y", 0))
        width, height = float(entity.get("width", 0)), float(entity.get("height", 0))
        if width <= 0 or height <= 0:
            raise ValueError("Cannot export an entity with a zero or negative size.")
        if str(entity.get("kind", "")).lower() == "circle":
            if abs(width - height) > 1e-6:
                raise ValueError("DXF circle entities must have equal width and height.")
            lines.extend((
                "0", "CIRCLE", "8", "SKETCH",
                "10", number(x + width / 2), "20", number(y + height / 2), "30", "0",
                "40", number(width / 2),
            ))
        elif str(entity.get("kind", "")).lower() == "rectangle":
            add_line(x, y, x + width, y)
            add_line(x + width, y, x + width, y + height)
            add_line(x + width, y + height, x, y + height)
            add_line(x, y + height, x, y)
        else:
            raise ValueError(f"Unsupported sketch entity: {entity.get('kind')}")
    lines.extend(("0", "ENDSEC", "0", "EOF", ""))
    return "\n".join(lines)


def import_dxf(text: str) -> list[dict]:
    """Read simple R12 CIRCLE and axis-aligned rectangle exports from this app."""
    raw_lines = [line.strip() for line in text.splitlines()]
    if len(raw_lines) % 2:
        raw_lines = raw_lines[:-1]
    pairs = []
    for index in range(0, len(raw_lines), 2):
        try:
            pairs.append((int(raw_lines[index]), raw_lines[index + 1]))
        except ValueError:
            continue
    entities, in_entities, current = [], False, None
    for code, value in pairs:
        if code == 0 and value == "SECTION":
            current = {"section": True}
        elif code == 2 and current and current.get("section"):
            in_entities = value == "ENTITIES"
            current = None
        elif not in_entities:
            continue
        elif code == 0:
            if current:
                entities.append(current)
            current = {"type": value}
        elif current is not None:
            current.setdefault("fields", {})[code] = value
    if current:
        entities.append(current)

    imported, line_groups = [], []
    rectangle_index = circle_index = 0
    for entity in entities:
        fields = entity.get("fields", {})
        if entity.get("type") == "CIRCLE":
            try:
                cx, cy, radius = float(fields[10]), float(fields[20]), float(fields[40])
            except (KeyError, ValueError):
                continue
            if radius > 0:
                circle_index += 1
                imported.append({
                    "kind": "circle", "name": f"Imported circle {circle_index}",
                    "x": cx - radius, "y": cy - radius, "width": radius * 2, "height": radius * 2,
                })
        elif entity.get("type") == "LINE":
            try:
                line_groups.append((
                    float(fields[10]), float(fields[20]), float(fields[11]), float(fields[21])
                ))
            except (KeyError, ValueError):
                continue

    # This lightweight importer recognizes four consecutive orthogonal lines as
    # one rectangle, matching the entities emitted by export_dxf().
    for offset in range(0, len(line_groups) - 3, 4):
        group = line_groups[offset:offset + 4]
        points = {(x, y) for x1, y1, x2, y2 in group for x, y in ((x1, y1), (x2, y2))}
        if len(points) != 4:
            continue
        xs = [point[0] for point in points]
        ys = [point[1] for point in points]
        left, right, top, bottom = min(xs), max(xs), min(ys), max(ys)
        corners = {(left, top), (right, top), (right, bottom), (left, bottom)}
        if points == corners and right > left and bottom > top:
            rectangle_index += 1
            imported.append({
                "kind": "rectangle", "name": f"Imported rectangle {rectangle_index}",
                "x": left, "y": top, "width": right - left, "height": bottom - top,
            })
    return imported

"""Approximate lead-screw actuator and motor sizing calculations."""
from __future__ import annotations

from dataclasses import dataclass
from math import pi


@dataclass(frozen=True)
class ScrewDriveResult:
    torque_nm: float
    rotational_speed_rpm: float
    mechanical_power_w: float
    input_power_w: float
    service_factor: float
    efficiency: float
    assumptions: tuple[str, ...]


class ActuatorInputError(ValueError):
    pass


def size_lead_screw_drive(
    axial_force_n: float,
    lead_mm_per_rev: float,
    linear_speed_mm_per_s: float,
    efficiency: float,
    service_factor: float = 1.5,
) -> ScrewDriveResult:
    """Estimate torque and speed for a screw drive under a steady axial load.

    Friction and screw losses are represented by one user-supplied efficiency.
    This does not select a motor, gearbox, driver, or safety-critical brake.
    """
    if axial_force_n <= 0:
        raise ActuatorInputError("Axial force must be positive.")
    if lead_mm_per_rev <= 0:
        raise ActuatorInputError("Screw lead must be positive.")
    if linear_speed_mm_per_s <= 0:
        raise ActuatorInputError("Target linear speed must be positive.")
    if not 0 < efficiency <= 1:
        raise ActuatorInputError("Efficiency must be greater than zero and at most 100%.")
    if service_factor < 1:
        raise ActuatorInputError("Use a service factor of at least 1.0.")

    torque = (axial_force_n * (lead_mm_per_rev / 1000.0) / (2 * pi * efficiency)) * service_factor
    rpm = (linear_speed_mm_per_s / lead_mm_per_rev) * 60
    mechanical_power = torque * 2 * pi * rpm / 60
    input_power = axial_force_n * (linear_speed_mm_per_s / 1000.0) / efficiency * service_factor
    return ScrewDriveResult(
        torque_nm=torque,
        rotational_speed_rpm=rpm,
        mechanical_power_w=mechanical_power,
        input_power_w=input_power,
        service_factor=service_factor,
        efficiency=efficiency,
        assumptions=(
            "Steady axial motion with a lead screw; acceleration and starting torque are omitted.",
            "Efficiency represents the screw drive only; motor and gearbox losses are omitted.",
            "Confirm peak load, duty cycle, thermal limits, buckling, critical speed, and brake needs separately.",
        ),
    )

"""Educational centroidal properties for common symmetric cross sections."""
from __future__ import annotations

from dataclasses import dataclass
from math import pi


@dataclass(frozen=True)
class SectionProperties:
    name: str
    area_mm2: float
    ixx_mm4: float
    iyy_mm4: float
    cxx_mm: float
    cyy_mm: float
    dimensions: dict[str, float]

    @property
    def section_modulus_x_mm3(self) -> float:
        return self.ixx_mm4 / self.cxx_mm

    @property
    def section_modulus_y_mm3(self) -> float:
        return self.iyy_mm4 / self.cyy_mm


class SectionInputError(ValueError):
    pass


def _positive(values: dict[str, float], names: tuple[str, ...]) -> None:
    missing = [name for name in names if name not in values]
    if missing:
        raise SectionInputError("Missing section dimensions: " + ", ".join(missing))
    invalid = [name for name in names if float(values[name]) <= 0]
    if invalid:
        raise SectionInputError("Dimensions must be greater than zero: " + ", ".join(invalid))


def calculate_section(shape: str, dimensions: dict[str, float]) -> SectionProperties:
    """Return A, Ixx, Iyy and extreme-fibre distances in mm-based units.

    Axes pass through the centroid. For rectangles and I-sections, x is the
    horizontal centroidal axis and y is the vertical centroidal axis.
    """
    shape = shape.strip().lower()
    d = {key: float(value) for key, value in dimensions.items()}
    if shape == "rectangle":
        _positive(d, ("width", "height"))
        width, height = d["width"], d["height"]
        area = width * height
        ixx = width * height**3 / 12
        iyy = height * width**3 / 12
        cxx, cyy = height / 2, width / 2
        name = "Rectangle"
    elif shape == "solid circle":
        _positive(d, ("diameter",))
        diameter = d["diameter"]
        area = pi * diameter**2 / 4
        ixx = iyy = pi * diameter**4 / 64
        cxx = cyy = diameter / 2
        name = "Solid circle"
    elif shape == "hollow circle":
        _positive(d, ("outer_diameter", "inner_diameter"))
        outer, inner = d["outer_diameter"], d["inner_diameter"]
        if inner >= outer:
            raise SectionInputError("Inner diameter must be smaller than outer diameter.")
        area = pi * (outer**2 - inner**2) / 4
        ixx = iyy = pi * (outer**4 - inner**4) / 64
        cxx = cyy = outer / 2
        name = "Hollow circle"
    elif shape == "rectangular tube":
        _positive(d, ("width", "height", "wall"))
        width, height, wall = d["width"], d["height"], d["wall"]
        if 2 * wall >= min(width, height):
            raise SectionInputError("Wall thickness must be less than half of the smaller outer dimension.")
        inner_width, inner_height = width - 2 * wall, height - 2 * wall
        area = width * height - inner_width * inner_height
        ixx = (width * height**3 - inner_width * inner_height**3) / 12
        iyy = (height * width**3 - inner_height * inner_width**3) / 12
        cxx, cyy = height / 2, width / 2
        name = "Rectangular tube"
    elif shape == "I-section":
        _positive(d, ("width", "height", "flange", "web"))
        width, height = d["width"], d["height"]
        flange, web = d["flange"], d["web"]
        if 2 * flange >= height:
            raise SectionInputError("Two flange thicknesses must fit within the total height.")
        if web >= width:
            raise SectionInputError("Web thickness must be smaller than the flange width.")
        clear_height = height - 2 * flange
        area = 2 * width * flange + web * clear_height
        ixx = (width * height**3 - (width - web) * clear_height**3) / 12
        iyy = (2 * flange * width**3 + clear_height * web**3) / 12
        cxx, cyy = height / 2, width / 2
        name = "I-section"
    else:
        raise SectionInputError(f"Unsupported shape: {shape}")

    if area <= 0 or ixx <= 0 or iyy <= 0:
        raise SectionInputError("Section dimensions do not form a valid shape.")
    return SectionProperties(name, area, ixx, iyy, cxx, cyy, d)

"""Elastic torsion response of solid and hollow circular shafts."""
from __future__ import annotations

from dataclasses import dataclass
from math import pi, sqrt


@dataclass(frozen=True)
class ShaftResult:
    polar_moment_mm4: float
    maximum_shear_stress_mpa: float
    twist_radians: float
    twist_degrees: float
    estimated_shear_yield_mpa: float | None
    yield_safety_factor: float | None


class TorsionInputError(ValueError):
    pass


def calculate_shaft_torsion(
    torque_nm: float,
    length_mm: float,
    shear_modulus_mpa: float,
    outer_diameter_mm: float,
    inner_diameter_mm: float = 0.0,
    tensile_yield_strength_mpa: float | None = None,
) -> ShaftResult:
    """Compute maximum elastic shear stress and angle of twist.

    Ductile-material shear yield is estimated from tensile yield by the
    von-Mises relation. This is a classroom estimate, not a fatigue check.
    """
    if min(torque_nm, length_mm, shear_modulus_mpa, outer_diameter_mm) <= 0:
        raise TorsionInputError("Torque, shaft length, shear modulus, and outer diameter must be positive.")
    if inner_diameter_mm < 0 or inner_diameter_mm >= outer_diameter_mm:
        raise TorsionInputError("Inner diameter must be zero or smaller than the outer diameter.")
    if tensile_yield_strength_mpa is not None and tensile_yield_strength_mpa <= 0:
        raise TorsionInputError("Tensile yield strength must be positive.")
    polar = pi * (outer_diameter_mm**4 - inner_diameter_mm**4) / 32
    torque_nmm = torque_nm * 1000
    stress = torque_nmm * (outer_diameter_mm / 2) / polar
    twist = torque_nmm * length_mm / (polar * shear_modulus_mpa)
    shear_yield = tensile_yield_strength_mpa / sqrt(3) if tensile_yield_strength_mpa else None
    safety = shear_yield / stress if shear_yield and stress > 0 else None
    return ShaftResult(
        polar_moment_mm4=polar,
        maximum_shear_stress_mpa=stress,
        twist_radians=twist,
        twist_degrees=twist * 180 / pi,
        estimated_shear_yield_mpa=shear_yield,
        yield_safety_factor=safety,
    )

class TrussSolver:
    """Solve a small, linear, planar truss in N and mm units.

    Coordinates and cross-sectional areas are in mm and mm². Young's modulus
    is in MPa (N/mm²), nodal loads in N, stress in MPa, and displacement in mm.
    Supports are fixed translations; each member carries only axial force.
    """

    @classmethod
    def solve(cls, nodes: list[Node], elements: list[TrussElement]) -> dict:
        if len(nodes) < 2:
            raise TrussInputError("Add at least two nodes before solving.")
        if not elements:
            raise TrussInputError("Add at least one truss member before solving.")

        dof_count = 2 * len(nodes)
        stiffness = np.zeros((dof_count, dof_count), dtype=float)
        force = np.zeros(dof_count, dtype=float)
        fixed_dofs: set[int] = set()

        for index, node in enumerate(nodes):
            force[2 * index] = node.force_x
            force[2 * index + 1] = node.force_y
            if node.fixed_x:
                fixed_dofs.add(2 * index)
            if node.fixed_y:
                fixed_dofs.add(2 * index + 1)
        if not fixed_dofs:
            raise TrussInputError("The structure has no supports. Fix at least one translation.")

        for element_index, element in enumerate(elements, start=1):
            if not (0 <= element.start < len(nodes) and 0 <= element.end < len(nodes)):
                raise TrussInputError(f"Member {element_index} refers to a node that does not exist.")
            if element.start == element.end:
                raise TrussInputError(f"Member {element_index} starts and ends at the same node.")
            if element.area <= 0 or element.youngs_modulus <= 0:
                raise TrussInputError(f"Member {element_index} needs positive area and modulus values.")

            first = nodes[element.start]
            second = nodes[element.end]
            dx, dy = second.x - first.x, second.y - first.y
            length = float(np.hypot(dx, dy))
            if length <= 1e-9:
                raise TrussInputError(f"Member {element_index} has zero length.")
            cosine, sine = dx / length, dy / length
            direction = np.array([-cosine, -sine, cosine, sine], dtype=float)
            member_matrix = (element.area * element.youngs_modulus / length) * np.outer(direction, direction)
            dofs = [2 * element.start, 2 * element.start + 1, 2 * element.end, 2 * element.end + 1]
            for row, global_row in enumerate(dofs):
                for column, global_column in enumerate(dofs):
                    stiffness[global_row, global_column] += member_matrix[row, column]

        free_dofs = [dof for dof in range(dof_count) if dof not in fixed_dofs]
        if not free_dofs:
            raise TrussInputError("Every degree of freedom is fixed; there is nothing to solve.")
        reduced = stiffness[np.ix_(free_dofs, free_dofs)]
        if np.linalg.matrix_rank(reduced, tol=1e-9) < len(free_dofs):
            raise TrussInputError(
                "The model is unstable or disconnected. Check the supports and member layout."
            )

        displacements = np.zeros(dof_count, dtype=float)
        try:
            displacements[free_dofs] = np.linalg.solve(reduced, force[free_dofs])
        except np.linalg.LinAlgError as exc:
            raise TrussInputError("The stiffness matrix could not be solved.") from exc
        reactions = stiffness @ displacements - force

        member_results = []
        for index, element in enumerate(elements, start=1):
            first, second = nodes[element.start], nodes[element.end]
            dx, dy = second.x - first.x, second.y - first.y
            length = float(np.hypot(dx, dy))
            cosine, sine = dx / length, dy / length
            dofs = [2 * element.start, 2 * element.start + 1, 2 * element.end, 2 * element.end + 1]
            extension = float(np.dot(np.array([-cosine, -sine, cosine, sine]), displacements[dofs]))
            stress = element.youngs_modulus * extension / length
            member_results.append({
                "member": index,
                "start_node": element.start + 1,
                "end_node": element.end + 1,
                "length_mm": length,
                "area_mm2": element.area,
                "axial_force_n": stress * element.area,
                "stress_mpa": stress,
                "state": "Tension" if stress >= 0 else "Compression",
            })

        node_results = []
        for index, node in enumerate(nodes):
            node_results.append({
                "node": index + 1,
                "x_mm": node.x,
                "y_mm": node.y,
                "displacement_x_mm": float(displacements[2 * index]),
                "displacement_y_mm": float(displacements[2 * index + 1]),
                "reaction_x_n": float(reactions[2 * index]) if node.fixed_x else 0.0,
                "reaction_y_n": float(reactions[2 * index + 1]) if node.fixed_y else 0.0,
            })
        return {
            "units": {"length": "mm", "force": "N", "stress": "MPa"},
            "members": member_results,
            "nodes": node_results,
            "maximum_displacement_mm": float(np.max(np.abs(displacements))),
            "maximum_absolute_stress_mpa": max(abs(row["stress_mpa"]) for row in member_results),
        }

"""Consistent engineering unit conversions for common project quantities."""
from __future__ import annotations

import math


UNIT_FACTORS = {
    "Length": {"mm": 1e-3, "cm": 1e-2, "m": 1.0, "in": 0.0254, "ft": 0.3048},
    "Area": {"mm²": 1e-6, "cm²": 1e-4, "m²": 1.0, "in²": 0.00064516},
    "Volume": {"mm³": 1e-9, "cm³": 1e-6, "m³": 1.0, "in³": 1.6387064e-5},
    "Force": {"N": 1.0, "kN": 1000.0, "lbf": 4.4482216153},
    "Torque": {"N·m": 1.0, "N·mm": 0.001, "lbf·ft": 1.3558179483},
    "Stress": {"Pa": 1.0, "kPa": 1000.0, "MPa": 1e6, "GPa": 1e9, "psi": 6894.7572932},
    "Rotational speed": {"rpm": 1.0, "rad/s": 60.0 / (2 * math.pi)},
    "Angle": {"degree": 1.0, "radian": 180.0 / math.pi},
    "Mass": {"kg": 1.0, "g": 0.001, "lb": 0.45359237},
}


def convert(value: float, category: str, source_unit: str, target_unit: str) -> float:
    """Convert a scalar through the category's canonical SI factor."""
    if not math.isfinite(value):
        raise ValueError("Enter a finite numeric value.")
    units = UNIT_FACTORS.get(category)
    if not units or source_unit not in units or target_unit not in units:
        raise ValueError("Choose matching units from one category.")
    return float(value) * units[source_unit] / units[target_unit]

class TrussPlot(QWidget):
    """Simple undeformed/deformed wire view to make the model and result intuitive."""

    def __init__(self, parent=None):
        super().__init__(parent)
        self.nodes: list[dict] = []
        self.elements: list[dict] = []
        self.results: dict | None = None
        self.setMinimumHeight(240)

    def set_model(self, nodes: list[dict], elements: list[dict], results: dict | None = None) -> None:
        self.nodes, self.elements, self.results = nodes, elements, results
        self.update()

    def paintEvent(self, event) -> None:
        painter = QPainter(self)
        painter.setRenderHint(QPainter.RenderHint.Antialiasing)
        painter.fillRect(self.rect(), QColor("#0d151c"))
        if not self.nodes:
            painter.setPen(QColor("#91a2b3"))
            painter.drawText(self.rect(), Qt.AlignmentFlag.AlignCenter, "Add nodes and members to preview the model")
            return
        xs = [float(row["x"]) for row in self.nodes]
        ys = [float(row["y"]) for row in self.nodes]
        dx, dy = max(max(xs) - min(xs), 1.0), max(max(ys) - min(ys), 1.0)
        pad = 38
        sx = (self.width() - 2 * pad) / dx
        sy = (self.height() - 2 * pad) / dy
        scale = min(sx, sy)
        origin_x = pad + (self.width() - 2 * pad - dx * scale) / 2
        origin_y = self.height() - pad - (self.height() - 2 * pad - dy * scale) / 2

        def point(x, y):
            return QPointF(origin_x + (x - min(xs)) * scale, origin_y - (y - min(ys)) * scale)

        displacements = self.results.get("nodes", []) if self.results else []
        max_disp = max((abs(r["displacement_x_mm"]) + abs(r["displacement_y_mm"]) for r in displacements), default=0)
        amplification = min(35.0, max(1.0, (dx + dy) * 0.06 / max(max_disp, 1e-9))) if self.results else 1.0
        for member in self.elements:
            a, b = int(member["start"]) - 1, int(member["end"]) - 1
            if 0 <= a < len(self.nodes) and 0 <= b < len(self.nodes):
                n1, n2 = self.nodes[a], self.nodes[b]
                painter.setPen(QPen(QColor("#617889"), 2))
                painter.drawLine(point(float(n1["x"]), float(n1["y"])), point(float(n2["x"]), float(n2["y"])))
                if self.results:
                    r1, r2 = displacements[a], displacements[b]
                    p1 = point(float(n1["x"]) + amplification * r1["displacement_x_mm"],
                               float(n1["y"]) + amplification * r1["displacement_y_mm"])
                    p2 = point(float(n2["x"]) + amplification * r2["displacement_x_mm"],
                               float(n2["y"]) + amplification * r2["displacement_y_mm"])
                    painter.setPen(QPen(QColor("#46d5a9"), 2, Qt.PenStyle.DashLine))
                    painter.drawLine(p1, p2)
        for index, node in enumerate(self.nodes):
            p = point(float(node["x"]), float(node["y"]))
            painter.setPen(QPen(QColor("#0d151c"), 1))
            painter.setBrush(QBrush(QColor("#f1b95e") if node["fixed_x"] or node["fixed_y"] else QColor("#7de0bd")))
            painter.drawEllipse(p, 6, 6)
            painter.setPen(QColor("#dbe5eb"))
            painter.drawText(p + QPointF(9, -7), str(index + 1))
        painter.setPen(QColor("#91a2b3"))
        legend = "Solid: original model"
        if self.results:
            legend += "     Dashed green: displaced shape (scaled)"
        painter.drawText(12, self.height() - 10, legend)

    @staticmethod
    def _table(headers: tuple[str, ...]) -> QTableWidget:
        table = QTableWidget(0, len(headers))
        table.setHorizontalHeaderLabels(headers)
        table.setSelectionBehavior(QAbstractItemView.SelectionBehavior.SelectRows)
        table.setSelectionMode(QAbstractItemView.SelectionMode.SingleSelection)
        table.horizontalHeader().setStretchLastSection(True)
        return table

    @staticmethod
    def _check_item(checked: bool) -> QTableWidgetItem:
        from PySide6.QtCore import Qt
        item = QTableWidgetItem()
        item.setFlags(Qt.ItemFlag.ItemIsEnabled | Qt.ItemFlag.ItemIsSelectable | Qt.ItemFlag.ItemIsUserCheckable)
        item.setCheckState(Qt.CheckState.Checked if checked else Qt.CheckState.Unchecked)
        return item

    def _load_sample(self) -> None:
        self.nodes.blockSignals(True)
        self.nodes.setRowCount(0)
        sample_nodes = [
            (0, 0, 0, 0, True, True),
            (400, 0, 0, 0, False, True),
            (400, 300, 0, -10000, False, False),
        ]
        for row, values in enumerate(sample_nodes):
            self.nodes.insertRow(row)
            for col, value in enumerate(values):
                self.nodes.setItem(row, col, self._check_item(bool(value)) if col >= 4 else QTableWidgetItem(str(value)))
        self.members.blockSignals(True)
        self.members.setRowCount(0)
        for row, values in enumerate(((1, 2, 300, 200000), (2, 3, 300, 200000), (1, 3, 300, 200000))):
            self.members.insertRow(row)
            for col, value in enumerate(values):
                self.members.setItem(row, col, QTableWidgetItem(str(value)))
        self.nodes.blockSignals(False)
        self.members.blockSignals(False)
        self._preview()
        self.output.clear()

    def _add_node(self) -> None:
        row = self.nodes.rowCount()
        self.nodes.insertRow(row)
        for col, value in enumerate((100 * row, 100, 0, 0)):
            self.nodes.setItem(row, col, QTableWidgetItem(str(value)))
        self.nodes.setItem(row, 4, self._check_item(False))
        self.nodes.setItem(row, 5, self._check_item(False))

    def _add_member(self) -> None:
        row = self.members.rowCount()
        self.members.insertRow(row)
        values = (1, min(2, max(1, self.nodes.rowCount())), 100, 200000)
        for col, value in enumerate(values):
            self.members.setItem(row, col, QTableWidgetItem(str(value)))

    @staticmethod
    def _remove_selected(table: QTableWidget) -> None:
        if table.currentRow() >= 0:
            table.removeRow(table.currentRow())

    @staticmethod
    def _text(table: QTableWidget, row: int, col: int) -> str:
        item = table.item(row, col)
        if item is None or not item.text().strip():
            raise TrussInputError(f"Fill in row {row + 1} of the input table.")
        return item.text().strip()

    def _read_model(self):
        nodes, model_nodes = [], []
        for row in range(self.nodes.rowCount()):
            x, y, fx, fy = (float(self._text(self.nodes, row, col)) for col in range(4))
            fixed_x = self.nodes.item(row, 4).checkState() == Qt.CheckState.Checked
            fixed_y = self.nodes.item(row, 5).checkState() == Qt.CheckState.Checked
            nodes.append(Node(x, y, fx, fy, fixed_x, fixed_y))
            model_nodes.append({"x": x, "y": y, "force_x": fx, "force_y": fy,
                                "fixed_x": fixed_x, "fixed_y": fixed_y})
        elements, model_elements = [], []
        for row in range(self.members.rowCount()):
            start = int(self._text(self.members, row, 0))
            end = int(self._text(self.members, row, 1))
            area, modulus = float(self._text(self.members, row, 2)), float(self._text(self.members, row, 3))
            elements.append(TrussElement(start - 1, end - 1, area, modulus))
            model_elements.append({"start": start, "end": end, "area_mm2": area,
                                   "youngs_modulus_mpa": modulus})
        return nodes, elements, {"nodes": model_nodes, "members": model_elements}

    def _preview(self) -> None:
        try:
            _, _, model = self._read_model()
            self.plot.set_model(model["nodes"], model["members"])
        except Exception:
            # Incomplete table edits should not interrupt typing.
            pass

    def refresh(self) -> None:
        project = self.get_active_project()
        self.project_label.setText(
            f"Analysis results will be saved to: {project['name']}" if project
            else "Choose a project before saving analysis results."
        )

    def _export_model(self) -> None:
        try:
            _, _, model = self._read_model()
        except Exception as exc:
            QMessageBox.warning(self, "Model has incomplete inputs", str(exc))
            return
        path, _ = QFileDialog.getSaveFileName(
            self, "Export truss model", "truss-model.json", "JSON model (*.json)"
        )
        if not path:
            return
        try:
            Path(path).with_suffix(".json").write_text(
                json.dumps({"units": {"length": "mm", "force": "N", "stress": "MPa"}, **model},
                           indent=2, ensure_ascii=False),
                encoding="utf-8",
            )
            QMessageBox.information(self, "Model exported", "Truss input saved as JSON.")
        except OSError as exc:
            QMessageBox.critical(self, "Could not export model", str(exc))

    def _import_model(self) -> None:
        path, _ = QFileDialog.getOpenFileName(self, "Import truss model", "", "JSON model (*.json)")
        if not path:
            return
        try:
            payload = json.loads(Path(path).read_text(encoding="utf-8"))
            nodes, members = payload["nodes"], payload["members"]
            if not isinstance(nodes, list) or not isinstance(members, list):
                raise ValueError("The JSON file needs nodes and members arrays.")
            self.nodes.blockSignals(True)
            self.members.blockSignals(True)
            self.nodes.setRowCount(0)
            self.members.setRowCount(0)
            for row, node in enumerate(nodes):
                self.nodes.insertRow(row)
                values = (node["x"], node["y"], node.get("force_x", 0), node.get("force_y", 0))
                for col, value in enumerate(values):
                    self.nodes.setItem(row, col, QTableWidgetItem(str(value)))
                self.nodes.setItem(row, 4, self._check_item(bool(node.get("fixed_x", False))))
                self.nodes.setItem(row, 5, self._check_item(bool(node.get("fixed_y", False))))
            for row, member in enumerate(members):
                self.members.insertRow(row)
                values = (
                    member["start"], member["end"], member.get("area_mm2", member.get("area")),
                    member.get("youngs_modulus_mpa", member.get("youngs_modulus")),
                )
                for col, value in enumerate(values):
                    self.members.setItem(row, col, QTableWidgetItem(str(value)))
            self.nodes.blockSignals(False)
            self.members.blockSignals(False)
            self.latest_results = None
            self.export_results_button.setEnabled(False)
            self.output.clear()
            self._preview()
        except (OSError, KeyError, TypeError, ValueError, json.JSONDecodeError) as exc:
            self.nodes.blockSignals(False)
            self.members.blockSignals(False)
            QMessageBox.critical(self, "Could not import model", str(exc))

    def _export_results_csv(self) -> None:

            with Path(path).with_suffix(".csv").open("w", newline="", encoding="utf-8-sig") as stream:
                writer = csv.writer(stream)

                writer.writerow(("Units", "length mm", "force N", "stress MPa"))
                writer.writerow(())
                writer.writerow(("Nodal results",))
                writer.writerow(("Node", "Ux mm", "Uy mm", "Reaction X N", "Reaction Y N"))
                for row in self.latest_results["nodes"]:
                    writer.writerow((row["node"], row["displacement_x_mm"], row["displacement_y_mm"],
                                     row["reaction_x_n"], row["reaction_y_n"]))
                writer.writerow(())
                writer.writerow(("Member results",))
                writer.writerow(("Member", "Start node", "End node", "State", "Axial force N", "Stress MPa"))
                for row in self.latest_results["members"]:
                    writer.writerow((row["member"], row["start_node"], row["end_node"], row["state"],
                                     row["axial_force_n"], row["stress_mpa"]))

        except OSError as exc:
            QMessageBox.critical(self, "Could not export CSV", str(exc))

    def _solve(self) -> None:
        try:
            nodes, elements, model = self._read_model()
            results = TrussSolver.solve(nodes, elements)
        except Exception as exc:
            QMessageBox.warning(self, "Analysis could not be completed", str(exc))
            return
        self.latest_results = results
        self.export_results_button.setEnabled(True)
        lines = [
            "LINEAR 2D TRUSS ANALYSIS", "Units: mm, N, MPa",
            f"Maximum displacement: {results['maximum_displacement_mm']:.6f} mm",
            f"Maximum absolute axial stress: {results['maximum_absolute_stress_mpa']:.3f} MPa",
            "", "NODAL RESULTS", "Node | Ux (mm) | Uy (mm) | Rx (N) | Ry (N)",
        ]
        for row in results["nodes"]:
            lines.append(
                f"{row['node']:>4} | {row['displacement_x_mm']:>10.5f} | {row['displacement_y_mm']:>10.5f} | "

def number_box(value: float, maximum: float = 1e9, decimals: int = 3) -> QDoubleSpinBox:
    box = QDoubleSpinBox()
    box.setRange(0, maximum)
    box.setDecimals(decimals)
    box.setValue(value)
    box.setSingleStep(max(0.1, value / 20))
    return box

    def _build_fields(self, shape: str) -> None:
        while self.form.rowCount():
            self.form.removeRow(0)
        self.fields.clear()
        for key, label, default in SHAPES[shape]:
            box = number_box(default, decimals=2)
            self.fields[key] = box
            self.form.addRow(label, box)

    def refresh(self) -> None:
        previous = self.material.currentData()
        self.material.clear()
        self.material.addItem("No material selected", None)
        if not self.database.connected:
            self.save_button.setEnabled(False)
            return
        try:
            for row in self.database.list_materials():
                self.material.addItem(row["name"], row)
            if previous:
                for index in range(self.material.count()):
                    data = self.material.itemData(index)
                    if data and data["name"] == previous["name"]:
                        self.material.setCurrentIndex(index)
                        break
        except Exception as exc:
            QMessageBox.warning(self, "Could not load material list", str(exc))
        self.save_button.setEnabled(self.get_active_project() is not None)

                self.shape.currentText(), {key: box.value() for key, box in self.fields.items()}
            )

            QMessageBox.warning(self, "Section dimensions need attention", str(exc))
            return
        material = self.material.currentData()

        mass_text = f"{self.result.area_mm2 * density * 1e-6:.4f} kg/m" if density else "Select a material to estimate mass"

            "iyy_mm4": self.result.iyy_mm4,
            "cxx_mm": self.result.cxx_mm,
            "cyy_mm": self.result.cyy_mm,
            "section_modulus_x_mm3": self.result.section_modulus_x_mm3,
            "section_modulus_y_mm3": self.result.section_modulus_y_mm3,
            "density_kg_m3": density,
            "material_name": material["name"] if material else None,
        }
        self.output.setPlainText(
            f"{self.result.name}\n"

            f"Extreme fibre distances cx / cy: {self.result.cxx_mm:.3f} / {self.result.cyy_mm:.3f} mm\n"
            f"Mass per metre: {mass_text}\n"
            "Axes pass through the centroid. Hollow profile overlaps are represented by subtraction."
        )
        self.section_calculated.emit({
            **self.result_values,

            "yield_strength_mpa": yield_strength,

class MotorResponsePlot(QWidget):
    """Lightweight position-versus-time chart drawn with Qt."""

    def __init__(self, parent=None):
        super().__init__(parent)
        self.time_s: list[float] = []
        self.position_deg: list[float] = []
        self.target_deg = 0.0
        self.setMinimumHeight(260)

    def set_response(self, simulation) -> None:
        self.time_s = simulation.time_s
        self.position_deg = simulation.position_deg
        self.target_deg = simulation.target_deg
        self.update()

    def paintEvent(self, event) -> None:
        painter = QPainter(self)
        painter.setRenderHint(QPainter.RenderHint.Antialiasing)
        painter.fillRect(self.rect(), QColor("#0d151c"))
        if not self.time_s:
            painter.setPen(QColor("#91a2b3"))
            painter.drawText(self.rect(), Qt.AlignmentFlag.AlignCenter, "Run the controller simulation to draw its response")
            return
        left, top, right, bottom = 52.0, 24.0, self.width() - 22.0, self.height() - 38.0
        times, values = self.time_s, self.position_deg
        t_max = max(times[-1], 1e-9)
        y_min = min(0.0, min(values))
        y_max = max(self.target_deg * 1.15, max(values) * 1.05, 1.0)
        if y_max - y_min < 1e-9:
            y_max = y_min + 1
        def point(index: int):
            x = left + (right - left) * times[index] / t_max
            y = bottom - (bottom - top) * (values[index] - y_min) / (y_max - y_min)
            return QPointF(x, y)
        painter.setPen(QPen(QColor("#304351"), 1))
        painter.drawRect(int(left), int(top), int(right - left), int(bottom - top))
        for tick in range(5):
            fraction = tick / 4
            y = bottom - fraction * (bottom - top)
            painter.setPen(QPen(QColor("#1c2b36"), 1))
            painter.drawLine(int(left), int(y), int(right), int(y))
            painter.setPen(QColor("#91a2b3"))
            label = y_min + fraction * (y_max - y_min)
            painter.drawText(4, int(y + 4), f"{label:.0f}°")
        target_y = bottom - (bottom - top) * (self.target_deg - y_min) / (y_max - y_min)
        painter.setPen(QPen(QColor("#e8b36a"), 1, Qt.PenStyle.DashLine))
        painter.drawLine(int(left), int(target_y), int(right), int(target_y))
        painter.setPen(QPen(QColor("#46d5a9"), 2))
        path = QPainterPath()
        path.moveTo(point(0))
        stride = max(1, len(times) // 1600)
        for index in range(stride, len(times), stride):
            path.lineTo(point(index))
        path.lineTo(point(len(times) - 1))
        painter.drawPath(path)
        painter.setPen(QColor("#91a2b3"))
        painter.drawText(int(left), self.height() - 10, "0 s")
        painter.drawText(int(right - 40), self.height() - 10, f"{t_max:.2f} s")
        painter.drawText(int(left + 8), int(top + 15), "Position response")

    def _add_stage(self) -> None:
        row = self.stages.rowCount()
        self.stages.insertRow(row)
        for col, value in enumerate((20, 60, 0.95)):
            self.stages.setItem(row, col, QTableWidgetItem(str(value)))

    def _remove_stage(self) -> None:
        if self.stages.currentRow() >= 0:
            self.stages.removeRow(self.stages.currentRow())

    def calculate(self) -> None:
        stages = []
        try:
            for row in range(self.stages.rowCount()):
                driver, driven, efficiency = (
                    self.stages.item(row, col).text().strip() if self.stages.item(row, col) else ""
                    for col in range(3)
                )
                if not driver or not driven or not efficiency:
                    raise GearInputError(f"Complete gear stage {row + 1}.")
                stages.append(GearStage(int(driver), int(driven), float(efficiency)))
            result = calculate_gear_train(self.input_torque.value(), self.input_speed.value(), stages)
        except (ValueError, GearInputError) as exc:
            QMessageBox.warning(self, "Gear inputs need attention", str(exc))
            return
        self.saved_result = asdict(result)
        self.output.setPlainText(
            f"Stages: {result.stages}\nTotal speed ratio: {result.total_ratio:.4f}:1 reduction\n"
            f"Combined efficiency: {result.total_efficiency:.1%}\n"
            f"Output speed: {result.output_speed_rpm:,.2f} rpm\n"
            f"Output torque estimate: {result.output_torque_nm:,.4f} N·m\n"
            f"Output mechanical power: {result.output_power_w:,.2f} W\n\n"
            "Direction changes once for each external gear mesh. Backlash, tooth strength, "
            "bearing losses, inertia, and gearbox ratings are not included."
        )

    def save(self) -> None:
        project = self.get_active_project()
        if not project or not self.database.connected:
            QMessageBox.information(self, "Project required", "Connect MySQL and select a project first.")
            return
        if not self.saved_result:
            QMessageBox.information(self, "Calculate first", "Calculate a gear train before saving it.")
            return
        rows = []
        for row in range(self.stages.rowCount()):
            rows.append([self.stages.item(row, col).text() for col in range(3)])
        inputs = {"input_torque_nm": self.input_torque.value(), "input_speed_rpm": self.input_speed.value(),
                  "stages": rows}
        try:
            name = "Gear train - " + datetime.now().strftime("%Y-%m-%d %H:%M:%S")
            self.database.save_design_check(project["id"], name, "gear_train", inputs, self.saved_result)
            QMessageBox.information(self, "Saved", "Gear calculation saved to the active project.")
        except Exception as exc:
            QMessageBox.critical(self, "Could not save gear calculation", str(exc))

            poisson = float(material["poisson_ratio"])
            self.shear_modulus.setValue(youngs / (2 * (1 + poisson)))

    def refresh(self) -> None:
        previous = self.material.currentData()
        self.material.clear()
        self.material.addItem("Custom shear modulus", None)
        if not self.database.connected:
            return
        try:
            for row in self.database.list_materials():
                self.material.addItem(row["name"], row)
            if previous:
                for index in range(self.material.count()):
                    data = self.material.itemData(index)
                    if data and data["name"] == previous["name"]:
                        self.material.setCurrentIndex(index)
                        break
        except Exception as exc:
            QMessageBox.warning(self, "Could not load material list", str(exc))

    def calculate(self) -> None:
        material = self.material.currentData()
        try:
            result = calculate_shaft_torsion(
                self.torque.value(), self.length.value(), self.shear_modulus.value(),
                self.outer.value(), self.inner.value() if self.shape.currentText() == "Hollow shaft" else 0.0,
                float(material["yield_strength_mpa"]) if material else None,
            )
        except (ValueError, TorsionInputError) as exc:
            QMessageBox.warning(self, "Shaft inputs need attention", str(exc))
            return
        self.result_values = asdict(result)
        factor = (
            f"{result.yield_safety_factor:.3f}" if result.yield_safety_factor is not None
            else "Add a material with yield strength for an estimate"
        )
        self.output.setPlainText(
            f"{self.shape.currentText()}\nPolar second moment J: {result.polar_moment_mm4:,.3f} mm⁴\n"
            f"Maximum elastic shear stress: {result.maximum_shear_stress_mpa:.3f} MPa\n"
            f"Angle of twist: {result.twist_radians:.6f} rad ({result.twist_degrees:.4f}°)\n"
            f"Estimated yield safety factor: {factor}\n"
            "Ductile shear yield is estimated from tensile yield with the von-Mises relation; fatigue and stress concentrations are omitted."
        )

    def save(self) -> None:
        project = self.get_active_project()
        if not project or not self.database.connected:
            QMessageBox.information(self, "Project required", "Connect MySQL and select a project first.")
            return
        if not self.result_values:
            QMessageBox.information(self, "Calculate first", "Calculate a shaft check before saving it.")
            return
        inputs = {
            "shape": self.shape.currentText(), "torque_nm": self.torque.value(),
            "length_mm": self.length.value(), "outer_diameter_mm": self.outer.value(),
            "inner_diameter_mm": self.inner.value() if self.shape.currentText() == "Hollow shaft" else 0,
            "shear_modulus_mpa": self.shear_modulus.value(),
            "material": self.material.currentText(),
        }
        try:
            label = "Shaft torsion - " + datetime.now().strftime("%Y-%m-%d %H:%M:%S")
            self.database.save_design_check(project["id"], label, "shaft_torsion", inputs, self.result_values)
            QMessageBox.information(self, "Saved", "Shaft check saved to the active project.")
        except Exception as exc:
            QMessageBox.critical(self, "Could not save shaft check", str(exc))

    def _set_units(self, category: str) -> None:
        units = tuple(UNIT_FACTORS[category].keys())
        current_source = self.source.currentText()
        current_target = self.target.currentText()
        self.source.blockSignals(True)
        self.target.blockSignals(True)
        self.source.clear()
        self.target.clear()
        self.source.addItems(units)
        self.target.addItems(units)
        if current_source in units:
            self.source.setCurrentText(current_source)
        if current_target in units:
            self.target.setCurrentText(current_target)
        elif len(units) > 1:
            self.target.setCurrentIndex(1)
        self.source.blockSignals(False)
        self.target.blockSignals(False)
        self.convert_value()

    def convert_value(self, *_args) -> None:
        if not self.source.currentText() or not self.target.currentText():
            return
        try:
            result = convert(
                self.value.value(), self.category.currentText(),
                self.source.currentText(), self.target.currentText(),
            )
            self.output.setPlainText(
                f"{self.value.value():,.8g} {self.source.currentText()} = "
                f"{result:,.10g} {self.target.currentText()}"
            )
        except ValueError as exc:
            self.output.setPlainText(str(exc))

        active = self.use_tls.isChecked()
        self.verify_tls.setEnabled(active)
        self.ssl_ca.setEnabled(active)
        self.browse_ca_button.setEnabled(active)

    def _browse_ca(self) -> None:
        path, _ = QFileDialog.getOpenFileName(
            self, "Choose MySQL server CA certificate", "",
            "Certificate files (*.pem *.crt *.cer);;All files (*)",
        )
        if path:
            self.ssl_ca.setText(path)

    @staticmethod
    def _is_local_host(host: str) -> bool:
        cleaned = host.strip().strip("[]").lower().rstrip(".")
        if (cleaned == "localhost" or cleaned.endswith(".localhost")
                or cleaned.startswith("127.") or cleaned.startswith("::ffff:127.")):
            return True
        try:
            return ipaddress.ip_address(cleaned).is_loopback
        except ValueError:
            return False

    @staticmethod
    def _is_unspecified_host(host: str) -> bool:
        cleaned = host.strip().strip("[]").lower().rstrip(".")
        try:
            return ipaddress.ip_address(cleaned).is_unspecified
        except ValueError:
            return cleaned in {"0", "0.0.0.0"}

    def _apply_local_defaults(self) -> None:
        if self._is_local_host(self.host.text()) and not self.ssl_ca.text().strip():
            self.use_tls.setChecked(False)
            self.verify_tls.setChecked(False)
        self._sync_tls_controls()

    def _connect(self) -> None:
        host = self.host.text().strip()
        database_name = self.database_name.text().strip()
        username = self.username.text().strip()
        password = self.password.text()
        if not host or not database_name or not username:
            QMessageBox.warning(self, "Missing connection details", "Enter the host, database name, and username.")
            return
        if self._is_unspecified_host(host):
            QMessageBox.warning(
                self, "Choose a MySQL host",
                "Enter localhost or a specific server name/IP. 0.0.0.0 is a bind address, not a connect address.",
            )
            return
        if not password:
            QMessageBox.warning(self, "Missing password", "Enter the password for the MySQL account.")
            return
        if self.use_tls.isChecked() and self.verify_tls.isChecked() and not self.ssl_ca.text().strip():
            QMessageBox.warning(
                self, "CA certificate required",
                "To verify the remote server certificate, select the CA certificate supplied by your MySQL provider. "
                "You can also turn off certificate verification if your provider instructs you to do so; TLS encryption will remain on.",
            )
            return

        self.connect_button.setEnabled(False)
        self.status.setText("Connecting to MySQL and setting up the project tables…")
        self.status.repaint()
        config = DatabaseConfig(
            host=host,
            port=self.port.value(),
            database=database_name,
            username=username,
            password=password,
            use_tls=self.use_tls.isChecked(),
            verify_tls=self.verify_tls.isChecked(),
            ssl_ca=self.ssl_ca.text().strip(),
        )
        try:
            self.database.connect(config)
        except Exception as exc:
            self.connect_button.setEnabled(True)
            self.status.setText("Connection was not established. Check that MySQL is running, the database exists, and the account/password are correct.")
            QMessageBox.critical(self, "MySQL connection failed", str(exc))
            return
        self.config = config
        self.remember_password = self.remember.isChecked()
        self.settings.save_connection(config, self.remember_password)
        self.accept()

        return self.active_project

    def _project_selected(self, project: dict | None) -> None:
        self.active_project = project
        for name in ("Overview", "2D Sketch", "Assembly", "Analysis", "Design Tools", "Reports"):
            page = self._pages.get(name)
            if name == "Overview":
                page.refresh()
            elif name == "2D Sketch":
                page.refresh()
            elif name == "Analysis":
                page.refresh()
            elif name == "Reports":
                page.refresh()
        self.statusBar().showMessage(
            f"Active project: {project['name']}" if project else "No active project", 5000
        )

    def navigate(self, name: str) -> None:
        if name not in self._pages:
            return
        self.stack.setCurrentWidget(self._pages[name])
        self.location.setText(name.upper())
        for page_name, button in self._nav_buttons.items():
            button.setChecked(page_name == name)
        page = self._pages[name]
        refresh = getattr(page, "refresh", None)
        if callable(refresh):
            refresh()

    def _restore_connection(self) -> None:
        saved = self.settings.load_connection()
        if not saved:
            self.configure_database()
            return
        config = DatabaseConfig(**saved)
        if not config.password:
            self.configure_database()
            return
        try:
            self.database.connect(config)
            self.settings.save_connection(config, remember_password=True)
            self._connection_changed(True)
        except Exception:
            self.configure_database()

    def configure_database(self) -> None:
        dialog = ConnectionDialog(self.database, self.settings, self)
        if dialog.exec() == ConnectionDialog.DialogCode.Accepted:
            self._connection_changed(True)
        elif not self.database.connected:
            self._connection_changed(False)

    def _connection_changed(self, connected: bool) -> None:
        if connected and self.database.config:
            self.connection_label.setText(f"●  Connected: {self.database.config.host}")
            self.connection_label.setStyleSheet("color:#7de0bd")
            self.statusBar().showMessage(
                f"Connected to MySQL database {self.database.config.database}.", 7000
            )
            self._pages["Projects"].refresh()
            # Project selection refreshes the dependent pages.
            if self.active_project is None:
                rows = self.database.list_projects()
                if rows:
                    self.active_project = {"id": int(rows[0]["id"]), "name": rows[0]["name"]}
            self._refresh_pages()
        else:
            self.connection_label.setText("●  MySQL not connected")
            self.connection_label.setStyleSheet("color:#e8b36a")
            self.statusBar().showMessage("Database is disconnected. You can reconnect from Database setup.", 7000)
            self.active_project = None
            self._refresh_pages()

    def _refresh_pages(self) -> None:
        for page in self._pages.values():
            refresh = getattr(page, "refresh", None)
            if callable(refresh):
                refresh()

    def closeEvent(self, event) -> None:
        self.database.disconnect()
        event.accept()

        layout.addWidget(subtitle)
        layout.addLayout(top)
        layout.addWidget(self.table, 1)

    def refresh(self) -> None:
        self.table.setRowCount(0)
        self.material_rows = {}
        if not self.database.connected:
            return
        try:
            rows = self.database.list_materials(self.search.text())
        except Exception as exc:
            QMessageBox.warning(self, "Could not load materials", str(exc))
            return
        for row in rows:
            index = self.table.rowCount()
            self.table.insertRow(index)
            self.material_rows[int(row["id"])] = row
            values = (
                row["id"], row["name"], row["category"],
                f"{float(row['youngs_modulus_mpa']):,.3f}",
                f"{float(row['poisson_ratio']):.4f}",
                f"{float(row['density_kg_m3']):,.1f}",
                f"{float(row['yield_strength_mpa']):,.2f}",
            )
            for col, value in enumerate(values):
                self.table.setItem(index, col, QTableWidgetItem(str(value)))

        selected = self.table.selectedItems()
        if not selected:
            QMessageBox.information(self, "Select a material", "Choose a material row to edit.")
            return
        row_index = selected[0].row()
        material_id = int(self.table.item(row_index, 0).text())
        material = self.material_rows.get(material_id)
        if material is None:
            return
        dialog = MaterialDialog(material, self)

            return
        name, category, modulus, poisson, density, yield_strength, notes = dialog.values()
        if not name or not category:

            return
        try:
            self.database.update_material(
                material_id, name, category, modulus, poisson, density, yield_strength, notes
            )
            self.refresh()
        except Exception as exc:
            QMessageBox.critical(self, "Could not update material", str(exc))

        if not self.database.connected:
            QMessageBox.information(self, "Connect database", "Configure the MySQL connection first.")
            return
        dialog = MaterialDialog(parent=self)
        if dialog.exec() != MaterialDialog.DialogCode.Accepted:
            return
        name, category, modulus, poisson, density, yield_strength, notes = dialog.values()
        if not name or not category:

            return
        try:
            self.database.add_material(name, category, modulus, poisson, density, yield_strength, notes)
            self.refresh()
        except Exception as exc:
            QMessageBox.critical(self, "Could not add material", str(exc))

        self.table.itemSelectionChanged.connect(self._selection_changed)

        buttons.addWidget(self.edit_button)
        buttons.addWidget(self.delete_button)
        buttons.addStretch(1)
        buttons.addWidget(self.refresh_button)
        layout = QVBoxLayout(self)
        layout.addWidget(title)
        layout.addWidget(subtitle)
        layout.addLayout(buttons)
        layout.addWidget(self.table, 1)

    def refresh(self, select_id: int | None = None) -> None:
        self.table.setRowCount(0)
        if not self.database.connected:
            return
        try:
            rows = self.database.list_projects()
        except Exception as exc:
            QMessageBox.warning(self, "Could not load projects", str(exc))
            return
        for row in rows:
            index = self.table.rowCount()
            self.table.insertRow(index)
            for col, value in enumerate((row["id"], row["name"], row["description"])):
                item = QTableWidgetItem(str(value))
                if col == 0:
                    item.setData(Qt.ItemDataRole.UserRole, int(row["id"]))
                self.table.setItem(index, col, item)
            if select_id is not None and int(row["id"]) == select_id:
                self.table.selectRow(index)
        if self.table.rowCount() and not self.table.selectedItems():
            self.table.selectRow(0)
        self._selection_changed()

    def _selection_changed(self) -> None:
        selected = self.table.selectedItems()
        if not selected:
            self.on_selected(None)
            return
        row = selected[0].row()
        project_id = int(self.table.item(row, 0).text())
        self.on_selected({"id": project_id, "name": self.table.item(row, 1).text()})

    def _create(self) -> None:
        if not self.database.connected:
            QMessageBox.information(self, "Connect database", "Configure the MySQL connection first.")
            return
        dialog = ProjectDialog(parent=self)
        if dialog.exec() != ProjectDialog.DialogCode.Accepted:
            return
        name, description = dialog.values()
        if not name:
            QMessageBox.warning(self, "Project name required", "Enter a name for this project.")
            return
        try:
            project_id = self.database.create_project(name, description)
            self.refresh(project_id)
        except Exception as exc:
            QMessageBox.critical(self, "Could not save project", str(exc))

    def _edit(self) -> None:
        selected = self.table.selectedItems()
        if not selected:
            return
        row = selected[0].row()
        project_id = int(self.table.item(row, 0).text())
        project = {
            "id": project_id,
            "name": self.table.item(row, 1).text(),
            "description": self.table.item(row, 2).text(),
        }
        dialog = ProjectDialog(project, self)
        if dialog.exec() != ProjectDialog.DialogCode.Accepted:
            return
        name, description = dialog.values()
        if not name:
            QMessageBox.warning(self, "Project name required", "Enter a name for this project.")
            return
        try:
            self.database.update_project(project_id, name, description)
            self.refresh(project_id)
        except Exception as exc:
            QMessageBox.critical(self, "Could not update project", str(exc))

    def _delete(self) -> None:
        selected = self.table.selectedItems()
        if not selected:
            return
        row = selected[0].row()
        project_id = int(self.table.item(row, 0).text())
        project_name = self.table.item(row, 1).text()
        answer = QMessageBox.question(
            self, "Delete project",
            f"Delete {project_name} and its sketches, BOM, and analysis records?",
        )
        if answer != QMessageBox.StandardButton.Yes:
            return
        try:
            self.database.delete_project(project_id)
            self.refresh()
        except Exception as exc:
            QMessageBox.critical(self, "Could not delete project", str(exc))

        try:

        self.preview.setPlainText(self._analysis_summary(row["name"], results))

    @staticmethod

        lines = [

            "",
            "Node results",
            "Node | Ux mm | Uy mm | Reaction X N | Reaction Y N",
        ]
        for row in results.get("nodes", []):
            lines.append(
                f"{row['node']} | {row['displacement_x_mm']:.6g} | {row['displacement_y_mm']:.6g} | "
                f"{row['reaction_x_n']:.6g} | {row['reaction_y_n']:.6g}"

            lines.append(
                f"{row['member']} | {row['state']} | {row['axial_force_n']:.6g} | {row['stress_mpa']:.6g}"
            )
        return "\n".join(lines)

"""Simple editable 2D parametric sketch page."""
from __future__ import annotations

from pathlib import Path

from PySide6.QtGui import QColor, QPen, QBrush, QPainter

class SketchScene(QGraphicsScene):
    def drawBackground(self, painter: QPainter, rect) -> None:
        painter.fillRect(rect, QColor("#0c141b"))
        grid = 25
        left = int(rect.left()) - (int(rect.left()) % grid)
        top = int(rect.top()) - (int(rect.top()) % grid)
        minor = QPen(QColor("#182631"), 0)
        major = QPen(QColor("#203542"), 0)
        x = left
        while x < rect.right():
            painter.setPen(major if x % (grid * 4) == 0 else minor)
            painter.drawLine(x, int(rect.top()), x, int(rect.bottom()))
            x += grid
        y = top
        while y < rect.bottom():
            painter.setPen(major if y % (grid * 4) == 0 else minor)
            painter.drawLine(int(rect.left()), y, int(rect.right()), y)
            y += grid

        self.rectangle_button.clicked.connect(lambda: self._add_entity("rectangle"))
        self.circle_button.clicked.connect(lambda: self._add_entity("circle"))

        self.save_button.clicked.connect(self.save)
        self.import_svg_button.clicked.connect(lambda: self._import_vector("svg"))
        self.import_dxf_button.clicked.connect(lambda: self._import_vector("dxf"))
        self.export_button.clicked.connect(self._export_svg)
        self.dxf_button.clicked.connect(self._export_dxf)
        self.scene.selectionChanged.connect(self._update_properties)
        toolbar = QHBoxLayout()
        toolbar.addWidget(self.rectangle_button)

        toolbar.addWidget(self.import_dxf_button)

    def _active(self):
        return self.get_active_project()

    def refresh(self) -> None:
        project = self._active()
        self.scene.clear()
        if not project:
            self.project_label.setText("Choose a project before drawing.")
            self._set_enabled(False)
            return
        self._set_enabled(self.database.connected)
        self.project_label.setText(f"Project: {project['name']}  •  drag shapes to position them")
        if not self.database.connected:
            return
        try:
            design = self.database.load_design(project["id"])
            if design:
                for entity in design.get("entities", []):
                    self._draw_entity(entity)
            self._update_properties()
        except Exception as exc:
            QMessageBox.warning(self, "Could not load sketch", str(exc))

    def _set_enabled(self, enabled: bool) -> None:
        for button in (self.rectangle_button, self.circle_button, self.delete_button, self.import_svg_button, self.import_dxf_button, self.export_button,
                            self.dxf_button, self.save_button):
            button.setEnabled(enabled)

    def _add_entity(self, kind: str) -> None:
        dialog = GeometryDialog(kind, self)
        if dialog.exec() == GeometryDialog.DialogCode.Accepted:
            self._draw_entity(dialog.values())

    def _draw_entity(self, entity: dict) -> None:
        x, y = float(entity.get("x", 0)), float(entity.get("y", 0))
        width, height = float(entity.get("width", 100)), float(entity.get("height", 100))
        pen = QPen(QColor("#61d6b0"), 2)
        brush = QBrush(QColor(48, 152, 125, 45))
        if entity.get("kind") == "circle":
            item = QGraphicsEllipseItem(0, 0, width, height)
        else:
            item = QGraphicsRectItem(0, 0, width, height)
        item.setPen(pen)
        item.setBrush(brush)
        item.setPos(x, y)
        item.setFlag(QGraphicsItem.GraphicsItemFlag.ItemIsSelectable, True)
        item.setFlag(QGraphicsItem.GraphicsItemFlag.ItemIsMovable, True)
        item.setData(0, dict(entity))
        self.scene.addItem(item)

    def _delete_selected(self) -> None:
        for item in self.scene.selectedItems():
            self.scene.removeItem(item)
        self._update_properties()

    def _update_properties(self) -> None:
        selected = self.scene.selectedItems()
        try:
            if selected:
                entity = selected[0].data(0)
                metrics = entity_metrics(entity)
                name = entity.get("name", "Profile")
                self.properties.setText(
                    f"Selected: {name} • {metrics['kind']} • area {metrics['area_mm2']:,.2f} mm² • "
                    f"perimeter {metrics['perimeter_mm']:,.2f} mm"
                )
            else:
                totals = sketch_totals(self._entities())
                self.properties.setText(
                    f"Sketch total: {int(totals['entity_count'])} entities • "
                    f"{totals['area_mm2']:,.2f} mm² profile area • "
                    f"{totals['perimeter_mm']:,.2f} mm perimeter"
                )
        except Exception as exc:
            self.properties.setText(str(exc))

    def _import_vector(self, kind: str) -> None:
        suffix = "svg" if kind == "svg" else "dxf"
        path, _ = QFileDialog.getOpenFileName(
            self, f"Import {suffix.upper()} geometry", "", f"{suffix.upper()} drawing (*.{suffix})"
        )
        if not path:
            return
        try:
            content = Path(path).read_text(encoding="utf-8-sig" if kind == "dxf" else "utf-8")
            entities = import_svg(content) if kind == "svg" else import_dxf(content)
            if not entities:
                QMessageBox.information(
                    self, "No supported shapes",
                    "This importer reads untransformed rectangles and circles. Other drawing entities are skipped.",
                )
                return
            for entity in entities:
                self._draw_entity(entity)
            self._update_properties()
            QMessageBox.information(self, "Geometry imported", f"Added {len(entities)} shapes to the current sketch.")
        except (OSError, ValueError, UnicodeError) as exc:
            QMessageBox.critical(self, "Could not import drawing", str(exc))

    def _export_svg(self) -> None:
        path, _ = QFileDialog.getSaveFileName(
            self, "Export sketch drawing", "MechaCAD-sketch.svg", "SVG drawing (*.svg)"
        )
        if not path:
            return
        try:
            content = export_svg(self._entities())
            Path(path).with_suffix(".svg").write_text(content, encoding="utf-8")
            QMessageBox.information(self, "SVG exported", f"Saved drawing to:\n{Path(path).with_suffix('.svg')}")
        except (OSError, ValueError) as exc:
            QMessageBox.critical(self, "Could not export sketch", str(exc))

    def _export_dxf(self) -> None:
        path, _ = QFileDialog.getSaveFileName(
            self, "Export sketch as DXF", "MechaCAD-sketch.dxf", "DXF drawing (*.dxf)"
        )
        if not path:
            return
        try:
            output = Path(path).with_suffix(".dxf")
            output.write_text(export_dxf(self._entities()), encoding="ascii")
            QMessageBox.information(self, "DXF exported", f"Saved drawing to:\n{output}")
        except (OSError, ValueError, UnicodeError) as exc:
            QMessageBox.critical(self, "Could not export DXF", str(exc))

    def _entities(self) -> list[dict]:
        entities = []
        for item in reversed(self.scene.items()):
            entity = item.data(0)
            if not isinstance(entity, dict):
                continue
            rect = item.rect()
            saved = dict(entity)
            saved.update({
                "x": round(item.pos().x(), 3),
                "y": round(item.pos().y(), 3),
                "width": round(rect.width(), 3),
                "height": round(rect.height(), 3),
            })
            entities.append(saved)
        return entities

    def save(self) -> None:
        project = self._active()
        if not project or not self.database.connected:
            QMessageBox.information(self, "Project required", "Connect MySQL and select a project first.")
            return
        try:
            self.database.save_design(project["id"], self._entities())
            QMessageBox.information(self, "Sketch saved", "The sketch is saved with this project.")
        except Exception as exc:
            QMessageBox.critical(self, "Could not save sketch", str(exc))
